    thick,

]

    % --- 1. 定义坐标点 ---

    \coordinate (C) at (4,2.5);

    % --- 2. 绘制梯形主体 ---
    \draw (A) -- (B) -- (C) -- (D) -- cycle;

    % 根据位置自动偏移，确保标签不与连线重合
    \node[left]  at (A) {$A$};

    \node[right] at (C) {$C$};
    \node[left]  at (D) {$D$};
