\documentclass[tikz,border=10pt]{standalone}
\begin{document}
\begin{tikzpicture}
    % Define coordinates based on a circle of radius 3
    % A and B are on the horizontal diameter
    \coordinate (A) at (-3, 0);
    \coordinate (B) at (3, 0);
    
    % C and D are on the circle such that CD is parallel to AB
    % Using coordinates derived from a 3-4-5 triangle relation for clean numbers:
    % x^2 + y^2 = 3^2. If y=2.4, x^2 = 9 - 5.76 = 3.24, so x = 1.8
    \coordinate (C) at (1.8, 2.4);
    \coordinate (D) at (-1.8, 2.4);
    
    % Draw the circle
    \draw (0,0) circle (3);
    
    % Draw the inscribed quadrilateral ABCD
    \draw (A) -- (B);
    \draw (B) -- (C);
    \draw (C) -- (D);
    \draw (D) -- (A);
    
    % Place labels
    \node[left] at (A) {$A$};
    \node[right] at (B) {$B$};
    \node[above right] at (C) {$C$};
    \node[above left] at (D) {$D$};
\end{tikzpicture}
\end{document}
