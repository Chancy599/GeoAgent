\documentclass[tikz,border=10pt]{standalone}
\begin{document}
\begin{tikzpicture}
    % Define coordinates
    \coordinate (O) at (0,0);
    \coordinate (M) at (-4,0);
    \coordinate (N) at (4,0);
    \coordinate (P) at (0,4);
    \coordinate (A) at (0,1.5);
    % Point B is on the circle. Let's place it at approx 35 degrees.
    % Radius is 4. x = 4*cos(35), y = 4*sin(35)
    \coordinate (B) at (3.28, 2.29);

    % Draw the semicircle arc
    \draw (M) arc (180:0:4);

    % Draw the diameter MN
    \draw (M) -- (N);

    % Draw the vertical radius OP
    \draw (O) -- (P);

    % Draw segment AB
    \draw (A) -- (B);

    % Draw segment OB
    \draw (O) -- (B);

    % Labels
    \node[below left] at (M) {$M$};
    \node[below right] at (N) {$N$};
    \node[below] at (O) {$O$};
    \node[above] at (P) {$P$};
    \node[left] at (A) {$A$};
    \node[right] at (B) {$B$};
\end{tikzpicture}
\end{document}
