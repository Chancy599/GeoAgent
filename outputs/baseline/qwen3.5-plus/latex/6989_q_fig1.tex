\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=1.2]
    % Define coordinates
    \coordinate (O) at (0,0);
    \coordinate (A) at (-3,0);
    \coordinate (B) at (3,0);
    % D is near the top, slightly right
    \coordinate (D) at (0.5, 2.96); 
    % C is in the bottom left quadrant
    \coordinate (C) at (-2.5, -1.66);

    % Draw the circle
    \draw (O) circle (3);

    % Draw lines
    \draw (A) -- (B);       % Diameter
    \draw (A) -- (D);       % Chord AD
    \draw (B) -- (D);       % Chord BD
    \draw (C) -- (D);       % Chord CD
    \draw (A) -- (C);       % Chord AC

    % Mark center O
    \fill (O) circle (1.5pt);

    % Labels
    \node[left] at (A) {$A$};
    \node[right] at (B) {$B$};
    \node[below left] at (C) {$C$};
    \node[above] at (D) {$D$};
    \node[below] at (O) {$O$};
\end{tikzpicture}
\end{document}
