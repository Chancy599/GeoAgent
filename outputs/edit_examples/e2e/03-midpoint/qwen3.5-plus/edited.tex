\documentclass{standalone}
\usepackage{tikz}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}
    % Define coordinates based on the visual appearance of the parallelogram
    \coordinate (B) at (0,0);
    \coordinate (C) at (5,0);
    \coordinate (A) at (1,2);
    \coordinate (D) at (6,2);

    % Draw the parallelogram
    \draw (A) -- (B) -- (C) -- (D) -- cycle;

    % Place original labels
    \node[above left] at (A) {$A$};
    \node[below left] at (B) {$B$};
    \node[below right] at (C) {$C$};
    \node[above right] at (D) {$D$};

    % Instruction: Add a solid dot at the midpoint of AB and label it M
    \coordinate (M) at ($(A)!0.5!(B)$);
    \fill (M) circle (2pt);
    \node[left] at (M) {$M$};
\end{tikzpicture}
\end{document}
