    % 坐标
    \coordinate (C) at (0,0);
    \coordinate (A) at (4,0);
    \coordinate (B) at (0,2);
    \coordinate (Ap) at (-2.1,4);
    \coordinate (Bp) at (-2.3,-1);
    \coordinate (D) at (-1.3,2.5); 
    \coordinate (E) at (-1.3,-0.6);

    \draw[thick] (A) -- (B) -- (C) -- cycle;
    \draw[thick] (Ap) -- (Bp) -- (C) -- cycle;

    \draw[semithick] (D) -- (E);
    \draw[semithick] (B) -- (E);

    % 标注
    \node[right] at (A) {$A$};
    \node[above] at (B) {$B$};
    \node[below right] at (C) {$C$};
    \node[above left] at (Ap) {$A'$};
    \node[left] at (Bp) {$B'$};
    \node[right] at (D) {$D$};
    \node[below] at (E) {$E$};
