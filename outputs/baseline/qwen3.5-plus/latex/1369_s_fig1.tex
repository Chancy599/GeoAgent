\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}
    % Define coordinates
    \coordinate (B) at (0,0);
    \coordinate (C) at (6,0);
    \coordinate (A) at (1,3);
    \coordinate (E) at (7,3);
    \coordinate (D) at (12,3);
    \coordinate (D') at (2,6);
    \coordinate (F) at (8,1);

    % Draw solid lines
    \draw (B) -- (C);
    \draw (C) -- (D);
    \draw (D) -- (E);
    \draw (E) -- (A);
    \draw (B) -- (D'); % Passes through A
    \draw (C) -- (D');
    \draw (D') -- (E);
    \draw (C) -- (E);
    \draw (A) -- (C);

    % Draw dashed line
    \draw[dashed] (E) -- (F);

    % Draw right angle mark at F
    % Vector FE is (-1, 2), Vector FC is (-2, -1)
    % We draw a small square corner
    \coordinate (F1) at ($(F)!0.25cm!(E)$);
    \coordinate (F2) at ($(F)!0.25cm!(C)$);
    \coordinate (F3) at ($(F1) + (F2) - (F)$);
    \draw (F1) -- (F3) -- (F2);

    % Labels
    \node[below left] at (B) {$B$};
    \node[below] at (C) {$C$};
    \node[right] at (D) {$D$};
    \node[above] at (D') {$D'$};
    \node[above] at (A) {$A$};
    \node[above] at (E) {$E$};
    \node[below right] at (F) {$F$};
\end{tikzpicture}
\end{document}
