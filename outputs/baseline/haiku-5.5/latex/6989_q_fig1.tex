\begin{tikzpicture}[scale=0.04, line width=0.8pt]

  \coordinate (A) at (-80,0);
  \coordinate (B) at (80,0);
  \coordinate (D) at (0,80);

  \node[left] at (A) {$A$};
  \node[right] at (B) {$B$};
  \node[below] at (O) {$O$};
  \node[below left] at (C) {$C$};
