\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=1.2]
  % Regular hexagon vertices
  \coordinate (A) at (0,0);
  \coordinate (B) at (2,0);
  \coordinate (C) at (3,1.732);
  \coordinate (D) at (2,3.464);
  \coordinate (E) at (0,3.464);
  \coordinate (F) at (-1,1.732);
  \coordinate (K) at (0,1.732);

  % Hexagon edges
  \draw (A) -- (B) -- (C) -- (D) -- (E) -- (F) -- (A);

  % Interior diagonals
  \draw (E) -- (A);
  \draw (F) -- (C);

  % Labels
  \node[above left] at (E) {$E$};
  \node[above right] at (D) {$D$};
  \node[right] at (C) {$C$};
  \node[below right] at (B) {$B$};
  \node[below left] at (A) {$A$};
  \node[left] at (F) {$F$};
  \node[below right] at (K) {$K$};
\end{tikzpicture}
\end{document}
