\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\begin{document}
\begin{tikzpicture}[scale=1]

% Coordinates
\coordinate (O) at (0,0.8);

\coordinate (C) at (-3.1,-0.55);

\coordinate (S) at (0,-0.05);

\draw (O) circle [radius=3.4];

% Triangle
\draw (A) -- (B) -- (C) -- cycle;

% Dashed segments
\draw[dashed] (O) -- (C);

\node[right] at (B) {$B$};
\node[left] at (C) {$C$};
