\documentclass[tikz,border=10pt]{standalone}
\usepackage{amsmath}

\begin{document}
\begin{tikzpicture}[scale=1.2, thick, line join=round, line cap=round]
    
    % --- 1. 坐标定义 ---
    % 按比例缩小绘制，每 10cm 对应 1 单位长度
    \coordinate (A) at (-2, 3);
    \coordinate (B) at (2, 3);
    \coordinate (C) at (-1, 2);
    \coordinate (D) at (1, 2);
    \coordinate (E) at (-1, 0);
    \coordinate (F) at (1, 0);

    % --- 2. 绘制轴截面轮廓 ---
    \draw (A) -- (B) -- (D) -- (F) -- (E) -- (C) -- cycle;

    % --- 3. 节点字母标定 ---
    \node[left] at (A) {$A$};
    \node[right] at (B) {$B$};
    \node[left] at (C) {$C$};
    \node[right] at (D) {$D$};
    \node[left] at (E) {$E$};
    \node[right] at (F) {$F$};

\end{tikzpicture}
\end{document}
