\documentclass[tikz,border=10pt]{standalone}
\begin{document}
\begin{tikzpicture}[line join=round,line cap=round,scale=1]

% Coordinates
\coordinate (A) at (0,0);
\coordinate (B) at (2.2,-2.6);
\coordinate (C) at (8.4,-2.6);
\coordinate (D) at (4.0,4.7);
\coordinate (O) at (3.55,0.0);

% Circle radius
\def\r{2.55}

% Outer quadrilateral edges
\draw (A)--(D);
\draw (D)--(C);
\draw (C)--(B);
\draw (B)--(A);

% Interior segments
\draw (A)--(O);
\draw (D)--(O);
\draw (B)--(O);
\draw (O)--(C);

% Incircle
\draw (O) circle (\r);

% Point O
\fill (O) circle (2.2pt);

% Labels
\node[left] at (A) {$A$};
\node[below] at (B) {$B$};
\node[below right] at (C) {$C$};
\node[above] at (D) {$D$};
\node[below right=1pt] at (O) {$O$};

\end{tikzpicture}
